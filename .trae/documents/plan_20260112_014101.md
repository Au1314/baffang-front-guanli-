#### 2.1.2 实现获取当前位置功能
添加`getCurrentLocation`函数，用于获取当前位置：
```javascript
// 获取当前位置
const getCurrentLocation = () => {
  if ('geolocation' in navigator) {
    navigator.geolocation.getCurrentPosition(
      (position) => {
        const longitude = position.coords.longitude
        const latitude = position.coords.latitude
        currentLocation.value = { longitude, latitude }
        locationError.value = null
        // 获取到位置后立即获取待响应呼叫列表
        fetchPendingEmergencyList()
      },
      (error) => {
        console.error('获取位置失败:', error.message)
        locationError.value = error.message
        ElMessage.warning('无法获取位置信息，请检查定位权限设置')
      },
      {
        enableHighAccuracy: true,
        timeout: 10000,
        maximumAge: 60000
      }
    )
  } else {
    locationError.value = '浏览器不支持地理定位'
    ElMessage.warning('浏览器不支持地理定位功能')
  }
}
```

#### 2.1.3 更新fetchPendingEmergencyList方法
修改现有的`fetchPendingEmergencyList`方法，使用真实API替代模拟数据：
```javascript
// 获取待处理紧急呼叫列表
const fetchPendingEmergencyList = async () => {
  try {
    emergencyListLoading.value = true
    if (!currentLocation.value) {
      // 如果还没有位置信息，先获取位置
      getCurrentLocation()
      return
    }
    
    const params = {
      pageNum: 1,
      pageSize: 10,
      longitude: currentLocation.value.longitude,
      latitude: currentLocation.value.latitude,
      type: 0, // 默认获取拖车类型，可根据需要调整
      maxDistance: 100, // 默认最大搜索距离100公里，可根据需要调整
      distanceType: 1 // 距离类型，1表示直线距离
    }
    
    const response = await authApi.getPendingEmergencyCalls(params)
    pendingEmergencyList.value = response.data.list || []
  } catch (error) {
    console.error('获取待处理紧急呼叫失败:', error)
    ElMessage.error('获取待处理紧急呼叫失败')
  } finally {
    emergencyListLoading.value = false
  }
}
```
